\begin{tikzpicture}[
                >={Stealth[length=2.5mm]},
                midarrow/.style={decoration={markings, mark=at position 0.45 with {\arrow{Stealth[length=2.5mm]}}}, postaction={decorate}},
                streamline/.style={thick, midarrow}
            ]
            
                % Solid Body (Wedge)
                \fill[pattern=north east lines, pattern color=gray!60] (0,0) -- (20:7.5) -- (-20:7.5) -- cycle;
                \draw[very thick] (0,0) -- (20:7.5);
                \draw[very thick] (0,0) -- (-20:7.5);
                
                % Symmetry Axis (s)
                \draw[thick, dashed, ->] (0,0) -- (8,0) node[right] {\Large $s$};
                
                % Angle Arc
                \draw[thick] (1.5, {1.5*sin(20)}) arc (20:-20:1.5);
                \node at (1.0, 0) {\Large $2\phi$};
                
                % Coordinate System & Boundary Layer (Rotated to align with top surface)
                \begin{scope}[rotate=20]
                    % x and y axes
                    \draw[thick, ->] (0,0) -- (7,0) node[below right] {\Large $x$};
                    \draw[thick, ->] (0,0) -- (0,3) node[left] {\Large $y$};
                    
                    % Boundary Layer Edge (Dashed curve)
                    \draw[thick, dashed] (0,0) .. controls (2,0.3) and (4,0.7) .. (7.5,1.2);
                    
                    % Velocity Profile U_infty(x)
                    \coordinate (Base) at (4.5,0);
                    \coordinate (BLEdge) at (4.5,0.83); % Height of BL at x=4.5
                    \coordinate (Top)  at (4.5,1.6);    % Total height of profile stack
                    
                    % Profile curve (curves inside BL, goes straight up outside)
                    \draw[thick] (Base) .. controls (4.7, 0.1) and (5.4, 0.4) .. (5.4, 0.83) -- (5.4, 1.6);
                    \draw (Base) -- (Top);
                    
                    % Velocity arrows INSIDE the boundary layer (growing)
                    \foreach \y in {0.15, 0.35, 0.55, 0.75} {
                        % Parabolic approximation for arrow length
                        \pgfmathsetmacro{\vLength}{0.9 * (1 - (1 - \y/0.83)^2)}
                        \draw[->] (4.5, \y) -- ++(\vLength, 0);
                    }
                    
                    % Velocity arrows OUTSIDE the boundary layer (constant free-stream)
                    \foreach \y in {0.95, 1.15, 1.35, 1.55} {
                        \draw[->] (4.5, \y) -- ++(0.9, 0);
                    }
                    
                    
                    
                    % Delta(x) Brace
                    \draw[decorate, decoration={brace, amplitude=6pt}, thick] 
                        (7.3, 1.15) -- (7.3, 0) node[midway, right=8pt] {\Large $\delta(x)$};
                \end{scope}
                
                % Free Stream Flow (U_FS)
                \node[left] at (-5.5, 0) {\huge $U_{FS}$};
                \foreach \y in {-1.5, -0.5, 0.5, 1.5} {
                    \draw[thick, ->] (-5.5, \y) -- (-4.5, \y);
                }
                
                % Streamlines
                % Top Streamlines
                \draw[streamline] (-4.5, 0.5) to[out=0, in=195] (-0.5, 1.3) to[out=18, in=200] (7.6,5.1);
                \draw[streamline] (-4.5, 1.5) to[out=0, in=190] (-0.5, 2.4) to[out=12, in=200] (7.6,6.1);
                
                % Bottom Streamlines
                \draw[streamline] (-4.5, -0.5) to[out=0, in=165] (-0.5, -1.1) to[out=-15, in=160] (7.6,-3.6);
                \draw[streamline] (-4.5, -1.5) to[out=0, in=170] (-0.5, -2.1) to[out=-10, in=160] (7.6,-4.6);
                
                % Streamline Label
                \node[left] at (-2.5, -3) {\Large streamlines};
                \draw[->, thick, bend left=15] (-2.7, -2.7) to (-1.8, -1.7);
                
            \begin{scope}[rotate=20]
\node[above, fill=white, inner sep=1pt] at (5.4, 1.65) {\Large $U_\infty(x)$};
\end{scope}
\end{tikzpicture}
