    \def\beltWidth{0.2}
    \def\filmThick{1.0}
    \def\beltHeight{5}
    \def\bulkLevel{0.5}
    % liquid
    \fill[cyan!15] (0, -0.5) -- (0, \beltHeight) -- (\filmThick, \beltHeight)
        -- (\filmThick, 1.5) to[out=-90, in=180] (3.5, \bulkLevel) -- (4.6, \bulkLevel) -- (4.6, -0.5) -- cycle;

    \fill[pattern=north east lines] (-\beltWidth, -0.5) rectangle (0, \beltHeight);

    \draw[thin] (-\beltWidth, -0.5) -- (-\beltWidth, \beltHeight);

    % free surface
    \draw[thick] (\filmThick, \beltHeight) -- (\filmThick, 1.5);
    \draw[thick] (\filmThick, 1.5) to[out=-90, in=180] (3.5, \bulkLevel) -- (4.6, \bulkLevel);

    \draw[->] (filmLabel) -- (0.6*\filmThick, 4.45);
    % axes
    \draw[thick, ->] (0, 2.6) -- (1.6, 2.6) node[right] {$y,\ v$};
